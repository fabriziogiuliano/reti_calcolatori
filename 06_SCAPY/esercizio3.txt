# 1. Definiamo la nostra destinazione.
destinazione = "1.1.1.1" # DNS di Cloudflare

# 2. Primo salto: creiamo un pacchetto con TTL=1.
# Il primo router sul percorso lo scarterà e ci risponderà.
pacchetto_hop1 = IP(dst=p, ttl=1) / UDP()

# 3. Inviamolo e vediamo chi ci risponde.
risposta_hop1 = sr1(pacchetto_hop1, timeout=2, verbose=0)

# 4. Stampiamo l'indirizzo IP del primo router (il nostro gateway).
# .src è l'IP sorgente del pacchetto di risposta.
print("Primo salto:", risposta_hop1.src)

# 5. Secondo salto: aumentiamo il TTL a 2.
pacchetto_hop2 = IP(dst=destinazione, ttl=2) / UDP()
risposta_hop2 = sr1(pacchetto_hop2, timeout=2, verbose=0)
print("Secondo salto:", risposta_hop2.src)

# 6. Terzo salto: TTL=3... e così via!
# L'IP della sorgente cambierà ad ogni hop, mappando il percorso.
pacchetto_hop3 = IP(dst=destinazione, ttl=3) / UDP()
risposta_hop3 = sr1(pacchetto_hop3, timeout=2, verbose=0)
print("Terzo salto:", risposta_hop3.src)
